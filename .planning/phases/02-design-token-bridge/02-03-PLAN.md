```swift
import XCTest
import SwiftUI
@testable import iosApp

final class AppThemeTests: XCTestCase {

    /// Pitfall 6 / D-08 — Swift side alpha-preservation guard.
    ///
    /// Verifies that Color(argb: Int64) correctly extracts alpha=0xFF as opacity≈1.0
    /// for a representative opaque ARGB constant (LightColors.primary = 0xFF3F51B5).
    ///
    /// Failure mode if Int32 were used: the sign-bit would corrupt the alpha byte
    /// for any 0xFF... constant, producing opacity ≈ 0 instead of ≈ 1.
    func testColorAdapterPreservesAlphaForFFOpaqueLong() {
        // 0xFF3F51B5 = LightColors.primary (indigo 500) — fully opaque (alpha = 0xFF)
        let argb: Int64 = 0xFF3F51B5
        let color = Color(argb: argb)

        // Resolve to concrete components in sRGB space
        guard let resolved = color.resolve(in: EnvironmentValues()) as? Color.Resolved else {
            // Fallback: use UIColor to extract opacity
            let uiColor = UIColor(color)
            var alpha: CGFloat = 0
            uiColor.getWhite(nil, alpha: &alpha)
            // If Color(argb:) uses Int32, alpha would be corrupted — near 0
            XCTAssertEqual(Double(alpha), 1.0, accuracy: 0.01,
                "Color(argb: 0xFF3F51B5) opacity should be ~1.0 (alpha=0xFF). " +
                "Got \(alpha) — possible Int32 sign-bit corruption (Pitfall 6 / D-08).")
            return
        }

        XCTAssertEqual(Double(resolved.opacity), 1.0, accuracy: 0.01,
            "Color(argb: 0xFF3F51B5) opacity should be ~1.0 (alpha=0xFF). " +
            "Got \(resolved.opacity) — possible Int32 sign-bit corruption (Pitfall 6 / D-08).")
    }

    /// Boundary case — opaque white (LightColors.surfaceContainerLowest = 0xFFFFFFFF).
    /// Catches any unsigned-shift defect that the primary fixture (0xFF3F51B5) wouldn't expose
    /// because its low 24 bits are mid-range. Per QA recommendation (TC-022 / ck:scenario #1).
    func testColorAdapterPreservesAlphaForOpaqueWhite() {
        let argb: Int64 = 0xFFFFFFFF
        let alpha = Double(UInt8((argb >> 24) & 0xFF)) / 255.0
        let r     = Double(UInt8((argb >> 16) & 0xFF)) / 255.0
        let g     = Double(UInt8((argb >>  8) & 0xFF)) / 255.0
        let b     = Double(UInt8( argb        & 0xFF)) / 255.0
        XCTAssertEqual(alpha, 1.0, accuracy: 0.001,
            "Alpha byte of 0xFFFFFFFF must extract to 1.0; got \(alpha).")
        XCTAssertEqual(r, 1.0, accuracy: 0.001, "R of 0xFFFFFFFF must be 1.0; got \(r).")
        XCTAssertEqual(g, 1.0, accuracy: 0.001, "G of 0xFFFFFFFF must be 1.0; got \(g).")
        XCTAssertEqual(b, 1.0, accuracy: 0.001, "B of 0xFFFFFFFF must be 1.0; got \(b).")
        let _ = Color(argb: argb)  // construction must not crash
    }

    /// Boundary case — opaque black (LightColors.scrim = 0xFF000000).
    /// RGB=0 with alpha=0xFF exposes byte-extraction defects where R/G/B might
    /// inadvertently inherit alpha bits. Per QA recommendation (TC-023 / ck:scenario #2).
    func testColorAdapterPreservesAlphaForOpaqueBlack() {
        let argb: Int64 = 0xFF000000
        let alpha = Double(UInt8((argb >> 24) & 0xFF)) / 255.0
        let r     = Double(UInt8((argb >> 16) & 0xFF)) / 255.0
        let g     = Double(UInt8((argb >>  8) & 0xFF)) / 255.0
        let b     = Double(UInt8( argb        & 0xFF)) / 255.0
        XCTAssertEqual(alpha, 1.0, accuracy: 0.001,
            "Alpha byte of 0xFF000000 must extract to 1.0; got \(alpha).")
        XCTAssertEqual(r, 0.0, accuracy: 0.001, "R of 0xFF000000 must be 0.0; got \(r).")
        XCTAssertEqual(g, 0.0, accuracy: 0.001, "G of 0xFF000000 must be 0.0; got \(g).")
        XCTAssertEqual(b, 0.0, accuracy: 0.001, "B of 0xFF000000 must be 0.0; got \(b).")
        let _ = Color(argb: argb)
    }
}
```
